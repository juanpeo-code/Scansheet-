Part 1: Google Apps Script Backend

Paste into Extensions > Apps Script

Appends row [Timestamp, Serial] sequentially from row 2 onwards: 
const SPREADSHEET_ID = ""; // Leave blank if script is bound via Extensions > Apps Script
const SHEET_NAME = "Scans";

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.tryLock(10000);

  try {
    let payload = {};
    if (e && e.postData && e.postData.contents) {
      try { payload = JSON.parse(e.postData.contents); } catch (err) { payload = e.parameter || {}; }
    } else if (e && e.parameter) {
      payload = e.parameter;
    }

    const serialNumber = (payload.serialNumber || payload.serial || "").toString().trim();
    if (!serialNumber) {
      return sendJsonResponse({ status: "error", message: "Missing 'serialNumber' in request." });
    }

    let ss = (SPREADSHEET_ID && SPREADSHEET_ID.trim()) 
      ? SpreadsheetApp.openById(SPREADSHEET_ID.trim()) 
      : SpreadsheetApp.getActiveSpreadsheet();

    if (!ss) {
      return sendJsonResponse({ status: "error", message: "Could not open Google Sheet." });
    }

    let sheet = ss.getSheetByName(SHEET_NAME) || ss.getSheets()[0];

    // Ensure Row 1 headers exist
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Timestamp", "Serial Number"]);
      sheet.getRange("A1:B1").setFontWeight("bold").setBackground("#f1f5f9");
    }

    const now = new Date();
    const timeZone = ss.getSpreadsheetTimeZone() || "GMT";
    const formattedTimestamp = Utilities.formatDate(now, timeZone, "yyyy-MM-dd HH:mm:ss");

    // Column A = Timestamp, Column B = Serial Number (preserves leading zeroes)
    sheet.appendRow([formattedTimestamp, "'" + serialNumber]);

    return sendJsonResponse({
      status: "success",
      message: "Scan appended successfully.",
      row: sheet.getLastRow(),
      timestamp: formattedTimestamp,
      serialNumber: serialNumber
    });

  } catch (err) {
    return sendJsonResponse({ status: "error", message: err.toString() });
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  return sendJsonResponse({ status: "ok", message: "ScanSheet Web App is live." });
}

function sendJsonResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}
Quick Deployment Checklist:
  1. Open Google Sheet > Extensions > Apps Script.
  2. Paste the code above and click Save.
  3. Click Deploy > New deployment > Select type: Web app.
  4. Set Execute as: Me and Who has access: Anyone.
  5. Copy the resulting Web App URL ending in /exec and paste into Settings.
